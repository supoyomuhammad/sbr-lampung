-- ============================================================================
-- ANOMALI USAHA SE2026 LAMPUNG — 20 flag
-- Blok A: kewajaran periode (berbasis rasio internal, bukan penempatan kolom)
-- ============================================================================
WITH params AS (
    SELECT '18' AS prov
),
asg AS (
    SELECT a.assignment_id, a.assignment_date_modified, a.survey_period_id,
           a.level_2_name AS kabupaten, a.level_3_name AS kecamatan,
           a.level_4_name AS desa, a.level_5_name AS sls,
           a.level_6_name AS sub_sls, a.level_6_full_code,
           CASE WHEN UPPER(a.code_identity) LIKE '%DTSEN%' THEN 'KELUARGA'
                WHEN TRIM(COALESCE(a.data6,'')) <> ''      THEN 'PRELIST USAHA'
                ELSE 'TAMBAHAN LAPANGAN' END AS jenis_asg,
           CONCAT('https://fasih-sm.bps.go.id/app/assignment/',
                  a.survey_period_id, '/', a.assignment_id) AS link_fasih
    FROM tgr_fd68e454.base_table_assignment a
    CROSS JOIN params p
    WHERE a.is_active = 1
      AND a.level_1_full_code = p.prov
      -- OPSI: AND a.level_2_full_code = '1871'
),
kel AS (
    SELECT r.assignment_id, r.assignment_date_modified, r.nama_kk
    FROM tgr_fd68e454.root_table r
),
usa AS (
    SELECT n.assignment_id, n.assignment_date_modified,
           n.nama_usaha, n.skala_usaha, n.kategori, n.kbli_akhir,
           n.badan_usaha_value, n.badan_usaha_label,
           n.jenis_usaha_label, n.lokasi_usaha_label, n.produk_sendiri_value,
           COALESCE(n.total_pendapatan, 0)      AS omset_thn,
           COALESCE(n.total_pendapatan_bln, 0)  AS omset_bln,
           COALESCE(n.total_pengeluaran, 0)     AS pengel_thn,
           COALESCE(n.total_pengeluaran_bln, 0) AS pengel_bln,
           COALESCE(n.gaji, 0)                  AS gaji_thn,
           COALESCE(n.gaji, n.gaji_bln, 0)      AS biaya_pekerja,
           COALESCE(n.biaya_produksi,    n.biaya_produksi_bln, 0)    AS r26b,
           COALESCE(n.total_pengeluaran, n.total_pengeluaran_bln, 0) AS r26f,
           CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) AS thn_operasi,
           CAST(COALESCE(n.tk_dibayar, 0)     AS SIGNED) AS pekerja_dibayar,
           CAST(COALESCE(n.tk_tdk_dibayar, 0) AS SIGNED) AS pekerja_tdk_dibayar,
           CAST(COALESCE(n.total_tk_jk, 0)    AS SIGNED) AS total_pekerja,
           CASE WHEN TRIM(COALESCE(n.kbli_akhir,'')) = ''
                 AND TRIM(COALESCE(n.kategori,'')) = ''
                 AND COALESCE(n.total_tk_jk,0) = 0
                 AND COALESCE(n.total_pendapatan, 0) = 0
                 AND COALESCE(n.total_pendapatan_bln, 0) = 0
                 AND TRIM(COALESCE(n.nama_usaha,'')) <> ''
                THEN 1 ELSE 0 END AS tagging_saja
    FROM tgr_fd68e454.se2026_nested n
    WHERE n.assignment_status_id > 0
      AND n.keberadaan_usaha_value IN ('1','2')
),
d AS (
    SELECT a.kabupaten, a.kecamatan, a.desa, a.sls, a.sub_sls,
           a.level_6_full_code, a.jenis_asg, a.link_fasih, k.nama_kk,
           u.nama_usaha, u.skala_usaha, u.kategori, u.kbli_akhir,
           u.badan_usaha_value, u.badan_usaha_label,
           u.jenis_usaha_label, u.lokasi_usaha_label, u.produk_sendiri_value,
           u.omset_thn, u.omset_bln, u.pengel_thn, u.pengel_bln, u.gaji_thn,
           u.biaya_pekerja, u.r26b, u.r26f, u.thn_operasi,
           u.pekerja_dibayar, u.pekerja_tdk_dibayar, u.total_pekerja,
           u.tagging_saja,
           CASE WHEN u.omset_thn > 0 THEN u.omset_thn
                ELSE u.omset_bln * 12 END AS omset_setahun
    FROM asg a
    INNER JOIN usa u ON u.assignment_id = a.assignment_id
                    AND u.assignment_date_modified = a.assignment_date_modified
    LEFT JOIN kel k  ON k.assignment_id = a.assignment_id
                    AND k.assignment_date_modified = a.assignment_date_modified
),
f AS (
    SELECT d.*,
           SUBSTR(CONCAT(
               -- A. KEWAJARAN PERIODE (rasio internal)
               CASE WHEN d.omset_thn > 0 AND d.pengel_thn > 0
                     AND d.omset_thn / d.pengel_thn BETWEEN 0.04 AND 0.20
                    THEN ' | A1 Omset thn setara nilai bulanan' ELSE '' END,
               CASE WHEN d.omset_bln > 0 AND d.pengel_bln > 0
                     AND d.omset_bln / d.pengel_bln BETWEEN 5 AND 20
                    THEN ' | A2 Omset bln setara nilai tahunan' ELSE '' END,
               CASE WHEN d.omset_thn > 0 AND d.gaji_thn > 0
                     AND d.omset_thn / d.gaji_thn BETWEEN 0.05 AND 0.20
                    THEN ' | A3 Omset thn kecil vs upah thn' ELSE '' END,
               CASE WHEN UPPER(TRIM(COALESCE(d.skala_usaha,''))) IN ('UM','UB')
                     AND d.total_pekerja > 0 AND d.omset_setahun > 0
                     AND d.omset_setahun / 12 / d.total_pekerja < 1000000
                    THEN ' | A4 Omset/pekerja/bln <1jt utk UM-UB' ELSE '' END,
               -- B. OMSET KECIL
               CASE WHEN d.omset_bln > 0 AND d.omset_bln < 50000
                    THEN ' | B1 Omset bln <50rb' ELSE '' END,
               CASE WHEN d.omset_bln >= 50000 AND d.omset_bln < 100000
                    THEN ' | B2 Omset bln 50-100rb' ELSE '' END,
               CASE WHEN d.omset_thn > 0 AND d.omset_thn < 500000
                    THEN ' | B3 Omset thn <500rb' ELSE '' END,
               CASE WHEN d.omset_thn >= 500000 AND d.omset_thn < 1000000
                    THEN ' | B4 Omset thn 500rb-1jt' ELSE '' END,
               -- C. OMSET BESAR
               CASE WHEN d.omset_bln > 4000000000 OR d.omset_thn > 50000000000
                    THEN ' | C1 Omset ekstrem tinggi' ELSE '' END,
               CASE WHEN d.total_pekerja < 5 AND d.omset_bln > 2000000000
                    THEN ' | C2 Omset bln >2M pekerja<5' ELSE '' END,
               CASE WHEN d.total_pekerja < 5 AND d.omset_thn > 15000000000
                    THEN ' | C3 Omset thn >15M pekerja<5' ELSE '' END,
               CASE WHEN d.total_pekerja >= 5 AND d.omset_bln > 50000000000
                    THEN ' | C4 Omset bln >50M' ELSE '' END,
               CASE WHEN d.total_pekerja >= 5 AND d.omset_thn > 500000000000
                    THEN ' | C5 Omset thn >500M' ELSE '' END,
               -- D. OMSET KOSONG & PEKERJA
               CASE WHEN d.omset_bln = 0 AND d.omset_thn = 0 AND d.tagging_saja = 0
                    THEN ' | D1 Omset kosong' ELSE '' END,
               CASE WHEN d.tagging_saja = 0 AND d.total_pekerja = 0
                     AND d.omset_setahun > 500000000
                    THEN ' | D2 Pekerja 0 omset >500jt' ELSE '' END,
               CASE WHEN d.total_pekerja > 5 AND d.omset_setahun < 10000000
                    THEN ' | D3 Pekerja >5 omset <10jt' ELSE '' END,
               CASE WHEN d.total_pekerja > 0
                     AND d.omset_setahun / NULLIF(d.total_pekerja,0) > 5000000000
                    THEN ' | D4 Produktivitas >5M/pekerja' ELSE '' END,
               CASE WHEN d.pekerja_dibayar > 0 AND d.biaya_pekerja = 0
                    THEN ' | D5 Pekerja dibayar tp gaji 0' ELSE '' END,
               CASE WHEN COALESCE(d.badan_usaha_value,'') NOT IN ('13','')
                     AND d.pekerja_tdk_dibayar > 0
                    THEN ' | D6 Badan hukum ada pekerja tdk dibayar' ELSE '' END,
               -- E. STRUKTUR BIAYA & LOKASI
               CASE WHEN d.produk_sendiri_value = '2'
                     AND d.r26f > 0 AND d.r26b / d.r26f > 0.5
                     AND UPPER(TRIM(COALESCE(d.kategori,'')))
                         NOT IN ('A','B','C','D','E','F','I')
                     AND LEFT(TRIM(COALESCE(d.kbli_akhir,'')),2)
                         NOT IN ('01','02','03')
                    THEN ' | E1 Biaya produksi dominan' ELSE '' END,
               CASE WHEN ( UPPER(TRIM(COALESCE(d.jenis_usaha_label,''))) LIKE '%KELILING%'
                        OR UPPER(TRIM(COALESCE(d.lokasi_usaha_label,''))) LIKE '%KELILING%' )
                     AND d.omset_setahun > 1000000000
                    THEN ' | E2 Keliling omset >1M/thn' ELSE '' END
           ), 4) AS flag_anomali
    FROM d
)
SELECT f.kabupaten, f.kecamatan, f.desa, f.sls, f.sub_sls,
       f.level_6_full_code AS id_sub_sls,
       f.nama_kk, f.nama_usaha, f.skala_usaha, f.kategori, f.kbli_akhir,
       f.jenis_usaha_label AS jenis_usaha,
       f.thn_operasi, f.total_pekerja, f.pekerja_dibayar,
       f.omset_thn, f.pengel_thn, f.omset_bln, f.pengel_bln, f.gaji_thn,
       CAST(f.omset_thn / NULLIF(f.pengel_thn,0) AS DECIMAL(18,3)) AS i1_rasio_thn,
       CAST(f.omset_bln / NULLIF(f.pengel_bln,0) AS DECIMAL(18,3)) AS i2_rasio_bln,
       CAST(f.omset_setahun / 12 / NULLIF(f.total_pekerja,0)
            AS DECIMAL(18,0)) AS i4_omset_pekerja_bln,
       f.flag_anomali,
       f.link_fasih
FROM f
WHERE f.flag_anomali <> ''
ORDER BY f.level_6_full_code, f.nama_usaha
LIMIT 9000 OFFSET 0;
