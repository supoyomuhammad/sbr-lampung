WITH params AS ( SELECT '18' AS prov ),
asg AS (
    SELECT a.assignment_id, a.assignment_date_modified, a.survey_period_id,
           a.level_2_name AS kabupaten, a.level_3_name AS kecamatan,
           a.level_4_name AS desa, a.level_5_name AS sls,
           a.level_6_name AS sub_sls, a.level_6_full_code,
           CONCAT('https://fasih-sm.bps.go.id/app/assignment/',
                  a.survey_period_id, '/', a.assignment_id) AS link_fasih
    FROM tgr_fd68e454.base_table_assignment a
    CROSS JOIN params p
    WHERE a.is_active = 1
      AND a.level_1_full_code = p.prov
),
u AS (
    SELECT n.assignment_id, n.assignment_date_modified,
           n.nama_usaha, n.kategori, n.kbli_akhir, n.keg_utama, n.skala_usaha,
           n.tahun_operasi,
           CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) AS thn_op,
           CAST(COALESCE(n.total_tk_jk, 0) AS SIGNED) AS total_tk,
           CAST(COALESCE(n.tk_dibayar, 0)  AS SIGNED) AS tk_dibayar,
           -- BLOK-AWARE: usaha 2026 mengisi R30-R32, usaha lama mengisi R26-R28
           CASE WHEN CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) = 2026
                THEN COALESCE(n.nilai_pendapatan_bln,0)
                ELSE COALESCE(n.nilai_pendapatan,0)  END AS r27a,
           CASE WHEN CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) = 2026
                THEN COALESCE(n.pendapatan_lain_bln,0)
                ELSE COALESCE(n.pendapatan_lain,0)   END AS r27b,
           CASE WHEN CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) = 2026
                THEN COALESCE(n.total_pendapatan_bln,0)
                ELSE COALESCE(n.total_pendapatan,0)  END AS r27c,
           CASE WHEN CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) = 2026
                THEN COALESCE(n.gaji_bln,0)
                ELSE COALESCE(n.gaji,0)              END AS r26a,
           CASE WHEN CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) = 2026
                THEN COALESCE(n.biaya_produksi_bln,0)
                ELSE COALESCE(n.biaya_produksi,0)    END AS r26b,
           CASE WHEN CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) = 2026
                THEN COALESCE(n.biaya_pembelian_bln,0)
                ELSE COALESCE(n.biaya_pembelian,0)   END AS r26c,
           CASE WHEN CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) = 2026
                THEN COALESCE(n.operasional_bln,0)
                ELSE COALESCE(n.operasional,0)       END AS r26d,
           CASE WHEN CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) = 2026
                THEN COALESCE(n.non_operasional_bln,0)
                ELSE COALESCE(n.non_operasional,0)   END AS r26e,
           CASE WHEN CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) = 2026
                THEN COALESCE(n.total_pengeluaran_bln,0)
                ELSE COALESCE(n.total_pengeluaran,0) END AS r26f,
           CASE WHEN CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) = 2026
                THEN COALESCE(n.nilai_pendapatan_bln,0) - COALESCE(n.biaya_pembelian_bln,0)
                ELSE COALESCE(n.nilai_pendapatan,0)     - COALESCE(n.biaya_pembelian,0)
           END AS nilai_output,
           CASE WHEN CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) = 2026
                THEN COALESCE(n.nilai_pendapatan_bln,0) - COALESCE(n.biaya_produksi_bln,0)
                   - COALESCE(n.biaya_pembelian_bln,0)  - COALESCE(n.operasional_bln,0)
                ELSE COALESCE(n.nilai_pendapatan,0)     - COALESCE(n.biaya_produksi,0)
                   - COALESCE(n.biaya_pembelian,0)      - COALESCE(n.operasional,0)
           END AS nilai_tambah
    FROM tgr_fd68e454.se2026_nested n
    WHERE n.assignment_status_id > 0
      AND n.keberadaan_usaha_value IN ('1','2')
),
f AS (
    SELECT a.kabupaten, a.kecamatan, a.desa, a.sls, a.sub_sls,
           a.level_6_full_code, a.link_fasih,
           u.assignment_id, u.nama_usaha, u.kategori, u.kbli_akhir,
           u.keg_utama, u.skala_usaha, u.tahun_operasi, u.thn_op,
           u.total_tk, u.tk_dibayar,
           u.r26a, u.r26b, u.r26c, u.r26d, u.r26e, u.r26f,
           u.r27a, u.r27b, u.r27c, u.nilai_output, u.nilai_tambah,
           SUBSTR(CONCAT(
               CASE WHEN u.total_tk > 0 AND u.nilai_output = 0
                     AND u.r26f > 0
                    THEN ' | F1 TK>0 tp output=0' ELSE '' END,
               CASE WHEN UPPER(TRIM(COALESCE(u.kategori,''))) IN ('P','U')
                    THEN ' | F2 Cakupan kategori P/U' ELSE '' END,
               CASE WHEN u.r26a > 0 AND u.nilai_tambah < u.r26a
                    THEN ' | F3 Nilai tambah < upah' ELSE '' END,
               CASE WHEN u.nilai_tambah < 0
                    THEN ' | F4 Nilai tambah negatif' ELSE '' END,
               CASE WHEN u.r27c > 0
                     AND ABS(u.r27c - (u.r27a + u.r27b)) > 0
                    THEN ' | F5 R27c <> R27a+R27b' ELSE '' END,
               CASE WHEN u.r26f > 0
                     AND ABS(u.r26f - (u.r26a + u.r26b + u.r26c + u.r26d + u.r26e)) > 0
                    THEN ' | F6 R26f <> jumlah rincian' ELSE '' END
           ), 4) AS flag_anomali
    FROM asg a
    INNER JOIN u ON u.assignment_id = a.assignment_id
                AND u.assignment_date_modified = a.assignment_date_modified
)
SELECT f.kabupaten, f.kecamatan, f.desa, f.sls, f.sub_sls,
       f.level_6_full_code AS id_sub_sls,
       f.nama_usaha, f.kategori, f.kbli_akhir,
       f.skala_usaha, f.tahun_operasi, f.total_tk, f.tk_dibayar,
       f.r26a AS upah_gaji, f.r26b AS biaya_produksi,
       f.r26c AS biaya_pembelian, f.r26d AS operasional,
       f.r26e AS non_operasional, f.r26f AS total_pengeluaran,
       f.r27a AS nilai_pendapatan, f.r27c AS total_pendapatan,
       f.nilai_output, f.nilai_tambah,
       f.flag_anomali,
       f.link_fasih
FROM f
WHERE f.flag_anomali <> ''
ORDER BY f.level_6_full_code, f.nama_usaha
LIMIT 9000 OFFSET 0;
