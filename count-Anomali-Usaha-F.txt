SELECT COUNT(*) AS total_baris
FROM tgr_fd68e454.base_table_assignment a
INNER JOIN tgr_fd68e454.se2026_nested n
        ON n.assignment_id = a.assignment_id
       AND n.assignment_date_modified = a.assignment_date_modified
WHERE a.is_active = 1
  AND a.level_1_full_code = '18'
  AND n.assignment_status_id > 0
  AND n.keberadaan_usaha_value IN ('1','2')
  AND ( -- F1
        ( CAST(COALESCE(n.total_tk_jk,0) AS SIGNED) > 0
          AND COALESCE(n.total_pengeluaran, n.total_pengeluaran_bln, 0) > 0
          AND COALESCE(n.nilai_pendapatan, n.nilai_pendapatan_bln, 0)
            - COALESCE(n.biaya_pembelian, n.biaya_pembelian_bln, 0) = 0 )
     -- F2
     OR UPPER(TRIM(COALESCE(n.kategori,''))) IN ('P','U')
     -- F3
     OR ( COALESCE(n.gaji, n.gaji_bln, 0) > 0
          AND COALESCE(n.nilai_pendapatan, n.nilai_pendapatan_bln, 0)
            - COALESCE(n.biaya_produksi,  n.biaya_produksi_bln, 0)
            - COALESCE(n.biaya_pembelian, n.biaya_pembelian_bln, 0)
            - COALESCE(n.operasional,     n.operasional_bln, 0)
            < COALESCE(n.gaji, n.gaji_bln, 0) )
     -- F4
     OR ( COALESCE(n.nilai_pendapatan, n.nilai_pendapatan_bln, 0)
        - COALESCE(n.biaya_produksi,  n.biaya_produksi_bln, 0)
        - COALESCE(n.biaya_pembelian, n.biaya_pembelian_bln, 0)
        - COALESCE(n.operasional,     n.operasional_bln, 0) < 0 )
     -- F5
     OR ( COALESCE(n.total_pendapatan, n.total_pendapatan_bln, 0) > 0
          AND ABS( COALESCE(n.total_pendapatan, n.total_pendapatan_bln, 0)
                 - (COALESCE(n.nilai_pendapatan, n.nilai_pendapatan_bln, 0)
                  + COALESCE(n.pendapatan_lain,  n.pendapatan_lain_bln, 0)) ) > 0 )
     -- F6
     OR ( COALESCE(n.total_pengeluaran, n.total_pengeluaran_bln, 0) > 0
          AND ABS( COALESCE(n.total_pengeluaran, n.total_pengeluaran_bln, 0)
                 - (COALESCE(n.gaji,            n.gaji_bln, 0)
                  + COALESCE(n.biaya_produksi,  n.biaya_produksi_bln, 0)
                  + COALESCE(n.biaya_pembelian, n.biaya_pembelian_bln, 0)
                  + COALESCE(n.operasional,     n.operasional_bln, 0)
                  + COALESCE(n.non_operasional, n.non_operasional_bln, 0)) ) > 0 ) );
