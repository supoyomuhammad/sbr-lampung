WITH params AS (
    SELECT '18' AS prov
),
asg AS (
    SELECT a.assignment_id,
           a.assignment_date_modified
    FROM tgr_fd68e454.base_table_assignment a
    CROSS JOIN params p
    WHERE a.is_active = 1
      AND a.level_1_full_code = p.prov
),
kel AS (
    SELECT r.assignment_id, r.assignment_date_modified
    FROM tgr_fd68e454.root_table r
),
usa AS (
    SELECT n.assignment_id,
           n.assignment_date_modified,
           n.badan_usaha_value,
           COALESCE(n.total_pendapatan, 0)     AS omset_thn,
           COALESCE(n.total_pendapatan_bln, 0) AS omset_bln,
           COALESCE(n.gaji, n.gaji_bln, 0)     AS biaya_pekerja,
           CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) AS thn_operasi,
           CAST(COALESCE(n.tk_dibayar, 0)     AS SIGNED) AS pekerja_dibayar,
           CAST(COALESCE(n.tk_tdk_dibayar, 0) AS SIGNED) AS pekerja_tdk_dibayar,
           CAST(COALESCE(n.total_tk_jk, 0)    AS SIGNED) AS total_pekerja,
           CASE WHEN TRIM(COALESCE(n.kbli_akhir,'')) = ''
                 AND TRIM(COALESCE(n.kategori,'')) = ''
                 AND COALESCE(n.total_tk_jk,0) = 0
                 AND COALESCE(n.total_pendapatan, 0) = 0
                 AND COALESCE(n.total_pendapatan_bln, 0) = 0
                 AND TRIM(COALESCE(n.nama_usaha,'')) <> ''
                THEN 1 ELSE 0 END AS tagging_saja
    FROM tgr_fd68e454.se2026_nested n
    WHERE n.assignment_status_id > 0
      AND n.keberadaan_usaha_value IN ('1','2')
),
deteksi AS (
    SELECT SUBSTR(CONCAT(
               CASE WHEN u.thn_operasi <= 2025 AND u.omset_thn = 0 AND u.omset_bln > 0 THEN ' | x' ELSE '' END,
               CASE WHEN u.thn_operasi = 2026 AND u.omset_thn > 0 THEN ' | x' ELSE '' END,
               CASE WHEN u.omset_bln > 0 AND u.omset_bln < 100000 THEN ' | x' ELSE '' END,
               CASE WHEN u.omset_thn > 0 AND u.omset_thn < 1000000 THEN ' | x' ELSE '' END,
               CASE WHEN u.omset_bln > 4000000000 OR u.omset_thn > 50000000000 THEN ' | x' ELSE '' END,
               CASE WHEN u.omset_bln = 0 AND u.omset_thn = 0 AND u.tagging_saja = 0 THEN ' | x' ELSE '' END,
               CASE WHEN u.tagging_saja = 0 AND u.total_pekerja = 0 
                     AND (CASE WHEN u.omset_thn > 0 THEN u.omset_thn ELSE u.omset_bln * 12 END) > 500000000 THEN ' | x' ELSE '' END,
               CASE WHEN u.total_pekerja > 5 
                     AND (CASE WHEN u.omset_thn > 0 THEN u.omset_thn ELSE u.omset_bln * 12 END) < 10000000 THEN ' | x' ELSE '' END,
               CASE WHEN u.total_pekerja > 0 
                     AND ((CASE WHEN u.omset_thn > 0 THEN u.omset_thn ELSE u.omset_bln * 12 END) / NULLIF(u.total_pekerja, 0)) > 5000000000 THEN ' | x' ELSE '' END,
               CASE WHEN u.pekerja_dibayar > 0 AND u.biaya_pekerja = 0 THEN ' | x' ELSE '' END,
               CASE WHEN COALESCE(u.badan_usaha_value,'') NOT IN ('13','') AND u.pekerja_tdk_dibayar > 0 THEN ' | x' ELSE '' END,
               CASE WHEN u.omset_bln > 0 AND u.omset_bln < 50000 THEN ' | x' ELSE '' END,
               CASE WHEN u.omset_thn > 0 AND u.omset_thn < 500000 THEN ' | x' ELSE '' END,
               CASE WHEN u.pekerja_dibayar < 5 AND u.omset_bln > 2000000000 THEN ' | x' ELSE '' END,
               CASE WHEN u.pekerja_dibayar < 5 AND u.omset_thn > 15000000000 THEN ' | x' ELSE '' END,
               CASE WHEN u.pekerja_dibayar >= 5 AND u.omset_bln > 50000000000 THEN ' | x' ELSE '' END,
               CASE WHEN u.pekerja_dibayar >= 5 AND u.omset_thn > 500000000000 THEN ' | x' ELSE '' END
           ), 4) AS flag_anomali
    FROM asg a
    INNER JOIN usa u ON u.assignment_id = a.assignment_id
                    AND u.assignment_date_modified = a.assignment_date_modified
    -- KEL CTE tidak perlu dipanggil di COUNT karena tidak mempengaruhi jumlah baris agregat
)
SELECT COUNT(*) AS total_baris_anomali
FROM deteksi d
WHERE d.flag_anomali <> '';
