SELECT COUNT(*) AS total_baris
FROM tgr_fd68e454.se2026_nested n
WHERE n.assignment_status_id > 0
  AND n.keberadaan_usaha_value IN ('1','2')
  AND ( ( CAST(NULLIF(TRIM(n.jenis_usaha_value),'') AS SIGNED) < 6
          AND UPPER(TRIM(COALESCE(n.kategori,''))) NOT IN ('B','C','D','E','F')
          AND n.produk_sendiri_value = '2'
          AND COALESCE(n.total_pengeluaran, n.total_pengeluaran_bln) > 0
          AND COALESCE(n.biaya_produksi, n.biaya_produksi_bln)
              / COALESCE(n.total_pengeluaran, n.total_pengeluaran_bln) > 0.5 )
     OR ( COALESCE(n.total_pendapatan, n.total_pendapatan_bln) IS NOT NULL
          AND COALESCE(n.total_pengeluaran, n.total_pengeluaran_bln) IS NOT NULL
          AND ( COALESCE(n.total_pendapatan, n.total_pendapatan_bln)
              - COALESCE(n.total_pengeluaran, n.total_pengeluaran_bln) ) < 0 )
     OR ( n.peran_mbg_value = '1'
          AND COALESCE(n.total_pengeluaran, n.total_pengeluaran_bln) > 0
          AND ( COALESCE(n.total_pendapatan, n.total_pendapatan_bln)
              / COALESCE(n.total_pengeluaran, n.total_pengeluaran_bln) >= 1.25
             OR COALESCE(n.total_pendapatan, n.total_pendapatan_bln)
              / COALESCE(n.total_pengeluaran, n.total_pengeluaran_bln) < 1 ) ) );
