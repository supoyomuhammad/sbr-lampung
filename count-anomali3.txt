SELECT COUNT(*) AS total_baris
FROM tgr_fd68e454.se2026_nested n
WHERE n.assignment_status_id > 0
  AND n.keberadaan_usaha_value IN ('1','2')
  AND COALESCE(n.total_aset_thn, n.total_aset_bln, 0) > 10000000000
  AND COALESCE(n.total_tk_jk, n.total_tk_bayar) = 1
  AND COALESCE(n.total_pendapatan, n.total_pendapatan_bln, 0) < 60000000;
