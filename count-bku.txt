WITH asg AS (
    SELECT a.assignment_id, a.assignment_date_modified, a.level_6_full_code,
           CASE WHEN UPPER(a.code_identity) LIKE '%DTSEN%' THEN 'KELUARGA'
                WHEN TRIM(COALESCE(a.data6,'')) <> ''      THEN 'PRELIST'
                ELSE 'TAMBAHAN' END AS kelompok
    FROM tgr_fd68e454.base_table_assignment a
    WHERE a.is_active = 1 --AND a.level_2_full_code = '1871'
),
usa AS (
    SELECT assignment_id, assignment_date_modified, COUNT(*) AS jml_usaha
    FROM tgr_fd68e454.se2026_nested
    --WHERE level_2_full_code = '1871'
    GROUP BY assignment_id, assignment_date_modified
)
SELECT COUNT(*) AS total_sub_sls
FROM (
    SELECT a.level_6_full_code
    FROM asg a
    LEFT JOIN usa u ON u.assignment_id = a.assignment_id
                   AND u.assignment_date_modified = a.assignment_date_modified
    WHERE a.kelompok = 'PRELIST'
       OR ( a.kelompok = 'TAMBAHAN' AND COALESCE(u.jml_usaha,0) > 0 )
    GROUP BY a.level_6_full_code
) x;
