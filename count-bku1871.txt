WITH asg AS (
    SELECT a.assignment_id, a.assignment_date_modified, a.assignment_status_alias AS status,
           CASE WHEN UPPER(a.code_identity) LIKE '%DTSEN%' THEN 'KELUARGA'
                WHEN TRIM(COALESCE(a.data6,'')) <> ''      THEN 'USAHA PRELIST'
                ELSE 'TAMBAHAN LAPANGAN' END AS kelompok
    FROM tgr_fd68e454.base_table_assignment a
    WHERE a.is_active = 1
      -- AND a.level_2_full_code = '1871'
),
usa AS (
    SELECT assignment_id, assignment_date_modified, COUNT(*) AS jml_usaha
    FROM tgr_fd68e454.se2026_nested
    -- WHERE level_2_full_code = '1871'
    GROUP BY assignment_id, assignment_date_modified
)
SELECT COUNT(*) AS total_baris
FROM asg a
LEFT JOIN usa u ON u.assignment_id = a.assignment_id
               AND u.assignment_date_modified = a.assignment_date_modified
WHERE a.kelompok = 'USAHA PRELIST'
   OR ( a.kelompok = 'TAMBAHAN LAPANGAN'
        AND COALESCE(u.jml_usaha,0) > 0 );
