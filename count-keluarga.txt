WITH params AS (
    SELECT '18' AS prov
)
SELECT COUNT(*) AS jml_assignment,
       CEIL(COUNT(*) / 24.0)          AS jml_kelompok,
       CEIL(COUNT(*) / 24.0 / 9000.0) AS jml_running,
       CAST(CEIL(COUNT(*) / 24.0) - 1 AS SIGNED) AS kelompok_terakhir
FROM tgr_fd68e454.base_table_assignment a
CROSS JOIN params p
INNER JOIN tgr_fd68e454.root_table r
        ON r.assignment_id = a.assignment_id
       AND r.assignment_date_modified = a.assignment_date_modified
WHERE a.is_active = 1
  AND a.level_1_full_code = p.prov
  AND ( LOWER(r.ada_keluarga_label) LIKE '%ditemukan%'
     OR LOWER(r.ada_keluarga_label) LIKE '%baru%'
     OR r.ada_keluarga_value IN ('1','2') )
  -- SALIN FILTER KABUPATEN YANG SAMA PERSIS DARI QUERY UTAMA KE SINI
  -- AND 
    a.level_2_full_code = '1871'
;
