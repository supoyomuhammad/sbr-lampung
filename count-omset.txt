SELECT COUNT(*) AS total_baris
FROM tgr_fd68e454.se2026_nested n
WHERE n.assignment_status_id > 0
  AND n.keberadaan_usaha_value IN ('1','2')
  AND ( COALESCE(n.total_pendapatan, n.total_pendapatan_bln) IS NULL
     OR COALESCE(n.total_pendapatan, n.total_pendapatan_bln) = 0
     OR ( COALESCE(n.total_pendapatan, n.total_pendapatan_bln) >= 1000000000
          AND n.total_tk_jk = 1 )
     OR ( CAST(NULLIF(TRIM(n.tahun_operasi),'') AS SIGNED) <= 2024
          AND n.total_pendapatan > 0 AND n.total_pengeluaran > 0
          AND n.total_pendapatan / n.total_pengeluaran BETWEEN 0.04 AND 0.16 )
     OR ( n.total_pendapatan IS NOT NULL AND n.nilai_pendapatan IS NOT NULL
          AND ABS(n.total_pendapatan
                  - (COALESCE(n.nilai_pendapatan,0) + COALESCE(n.pendapatan_lain,0))) > 1 )
     OR COALESCE(n.total_pendapatan, n.total_pendapatan_bln) >= 1000000000000
     OR ( n.total_pendapatan IS NOT NULL AND n.total_pengeluaran IS NOT NULL
          AND n.total_pendapatan = n.total_pengeluaran AND n.total_pendapatan > 0 ) );
