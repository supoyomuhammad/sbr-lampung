WITH params AS (
    SELECT CAST('2026-07-19' AS DATE) AS tgl_mulai,
           CAST('2026-07-28' AS DATE) AS tgl_akhir
),
first_event AS (
    SELECT h.assignment_id_timestamp, h.user_id, h.status_alias, h.level_2_full_code,
           CAST(DATE_ADD(h.date_created, INTERVAL 7 HOUR) AS DATE) AS tgl_wib,
           ROW_NUMBER() OVER (
               PARTITION BY h.assignment_id_timestamp,
                            CASE WHEN h.status_alias = 'SUBMITTED BY Pencacah' THEN 'SUBMIT'
                                 ELSE 'REVIEW' END
               ORDER BY h.date_created ASC, h.user_id ASC
           ) AS rn
    FROM tgr_fd68e454.base_table_assignment_history h
    WHERE h.status_alias IN ('SUBMITTED BY Pencacah', 'APPROVED BY Pengawas', 'REJECTED BY Pengawas')
),
aktivitas_harian AS (
    SELECT e.level_2_full_code, e.user_id, e.tgl_wib,
           SUM(CASE WHEN e.status_alias = 'SUBMITTED BY Pencacah' THEN 1 ELSE 0 END) AS progres_pcl,
           SUM(CASE WHEN e.status_alias IN ('APPROVED BY Pengawas','REJECTED BY Pengawas') THEN 1 ELSE 0 END) AS progres_pml
    FROM first_event e
    CROSS JOIN params p
    WHERE e.rn = 1 
      AND e.tgl_wib BETWEEN p.tgl_mulai AND p.tgl_akhir
    GROUP BY e.level_2_full_code, e.user_id, e.tgl_wib
)
SELECT COUNT(1) AS jml_baris
FROM aktivitas_harian;
