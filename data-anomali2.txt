SELECT n.level_2_name AS kabupaten, n.level_3_name AS kecamatan,
       n.level_4_name AS desa_kel, n.level_5_name AS sls,
       n.level_6_name AS sub_sls, n.level_6_full_code AS id_sub_sls,
       n.nama_usaha, n.assignment_id,
       '3.Penyertaan Modal Korporasi' AS flag_anomali,
       n.idsbr, n.skala_usaha,
       n.badan_usaha_value AS r11a_kode,
       n.badan_usaha_label AS r11a_badan_usaha,
       n.pribadi     AS r29a_pribadi,
       n.non_profit  AS r29b_nonprofit,
       n.publik      AS r29c_korp_publik,
       n.non_publik  AS r29d_korp_nonpublik,
       n.pemerintah  AS r29e_pemerintah,
       n.asing       AS r29f_asing,
       n.info_total  AS r29g_total,
       n.publik_didirikan    AS r33c_korp_publik,
       n.nonpublik_didirikan AS r33d_korp_nonpublik,
       CONCAT('https://fasih-sm.bps.go.id/app/assignment/',
              n.survey_period_id,'/',n.assignment_id) AS link_fasih
FROM tgr_fd68e454.se2026_nested n
WHERE n.assignment_status_id > 0
  AND n.keberadaan_usaha_value IN ('1','2')
  AND n.badan_usaha_value = '13'
  AND ( COALESCE(n.publik,0) > 0 OR COALESCE(n.non_publik,0) > 0
     OR COALESCE(n.publik_didirikan,0) > 0 OR COALESCE(n.nonpublik_didirikan,0) > 0 )
ORDER BY n.level_6_full_code, n.assignment_id
LIMIT 9000 OFFSET 0;
