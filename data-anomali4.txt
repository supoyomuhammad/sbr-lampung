WITH f AS (
    SELECT n.assignment_id, n.survey_period_id,
           n.level_2_name, n.level_3_name, n.level_4_name,
           n.level_5_name, n.level_6_name, n.level_6_full_code,
           n.nama_usaha, n.idsbr, n.skala_usaha,
           n.internet_label, n.lap_keuangan_label, n.badan_usaha_label,
           n.kbli_prelist, n.kbli_akhir, n.kategori, n.kategori_2025,
           n.tahun_operasi, n.total_tk_jk,
           COALESCE(n.total_pendapatan, n.total_pendapatan_bln) AS r27c,
           CASE WHEN UPPER(TRIM(COALESCE(n.skala_usaha,''))) IN ('UM','UB')
                 AND n.internet_value = '2'     THEN 1 ELSE 0 END AS a6,
           CASE WHEN UPPER(TRIM(COALESCE(n.skala_usaha,''))) IN ('UM','UB')
                 AND n.lap_keuangan_value = '2' THEN 1 ELSE 0 END AS a7,
           CASE WHEN TRIM(COALESCE(n.kbli_prelist,'')) <> ''
                 AND TRIM(COALESCE(n.kbli_akhir,''))   <> ''
                 AND LEFT(TRIM(n.kbli_prelist),2) <> LEFT(TRIM(n.kbli_akhir),2)
                                                 THEN 1 ELSE 0 END AS a8
    FROM tgr_fd68e454.se2026_nested n
    WHERE n.assignment_status_id > 0
      AND n.keberadaan_usaha_value IN ('1','2')
)
SELECT f.level_2_name AS kabupaten, f.level_3_name AS kecamatan,
       f.level_4_name AS desa_kel, f.level_5_name AS sls,
       f.level_6_name AS sub_sls, f.level_6_full_code AS id_sub_sls,
       f.nama_usaha, f.assignment_id,
       SUBSTR(CONCAT(
          CASE WHEN f.a6=1 THEN ' | 6.Internet UM-UB'      ELSE '' END,
          CASE WHEN f.a7=1 THEN ' | 7.Lap.Keuangan UM-UB'  ELSE '' END,
          CASE WHEN f.a8=1 THEN ' | 8.Beda KBLI vs SBR'    ELSE '' END), 4) AS flag_anomali,
       f.idsbr, f.skala_usaha, f.r27c AS r27c_pendapatan,
       f.total_tk_jk AS r24c1_pekerja, f.tahun_operasi,
       f.internet_label      AS r16a_internet,
       f.lap_keuangan_label  AS r11d_lap_keuangan,
       f.badan_usaha_label   AS r11a_badan_usaha,
       f.kbli_prelist, f.kbli_akhir,
       LEFT(TRIM(f.kbli_prelist),2) AS kbli2_sbr,
       LEFT(TRIM(f.kbli_akhir),2)   AS kbli2_hasil,
       f.kategori, f.kategori_2025,
       CONCAT('https://fasih-sm.bps.go.id/app/assignment/',
              f.survey_period_id,'/',f.assignment_id) AS link_fasih
FROM f
WHERE f.a6 + f.a7 + f.a8 > 0
ORDER BY f.level_6_full_code, f.assignment_id
LIMIT 1000 OFFSET 0;
