-- ============================================================================
-- PEMADATAN BARIS: 24 assignment per baris, pemisah ^
-- Kapasitas sekali jalan: 9.000 baris x 24 = 216.000 assignment
-- ============================================================================
WITH params AS (
    SELECT '18' AS prov
),
base AS (
    SELECT a.assignment_id,
           a.level_2_full_code, a.level_2_name,
           a.level_3_name, a.level_4_name, a.level_6_full_code,
           r.geotag_latitude, r.geotag_longitude,
           r.nama_kk, r.nama_ak_lain, r.ada_keluarga_label, a.data1
    FROM tgr_fd68e454.base_table_assignment a
    CROSS JOIN params p
    INNER JOIN tgr_fd68e454.root_table r
            ON r.assignment_id = a.assignment_id
           AND r.assignment_date_modified = a.assignment_date_modified
    WHERE a.is_active = 1
      AND a.level_1_full_code = p.prov
      AND ( LOWER(r.ada_keluarga_label) LIKE '%ditemukan%'
         OR LOWER(r.ada_keluarga_label) LIKE '%baru%'
         OR r.ada_keluarga_value IN ('1','2') )
      -- ==================== OPSI FILTER KABUPATEN/KOTA ====================
      -- Aktifkan SALAH SATU baris di bawah ini
      AND a.level_2_full_code = '1801'   -- Lampung Barat
      -- AND a.level_2_full_code = '1802'   -- Tanggamus
      -- AND a.level_2_full_code = '1803'   -- Lampung Selatan
      -- AND a.level_2_full_code = '1804'   -- Lampung Timur
      -- AND a.level_2_full_code = '1805'   -- Lampung Tengah
      -- AND a.level_2_full_code = '1806'   -- Lampung Utara
      -- AND a.level_2_full_code = '1807'   -- Way Kanan
      -- AND a.level_2_full_code = '1808'   -- Tulang Bawang
      -- AND a.level_2_full_code = '1809'   -- Pesawaran
      -- AND a.level_2_full_code = '1810'   -- Pringsewu
      -- AND a.level_2_full_code = '1811'   -- Mesuji
      -- AND a.level_2_full_code = '1812'   -- Tulang Bawang Barat
      -- AND a.level_2_full_code = '1813'   -- Pesisir Barat
      -- AND a.level_2_full_code = '1871'   -- Bandar Lampung
      -- AND a.level_2_full_code = '1872'   -- Metro
      -- ---- gabungan beberapa kabupaten ----
      -- AND a.level_2_full_code IN ('1804','1805')          -- dua terbesar
      -- AND a.level_2_full_code IN ('1871','1872')          -- perkotaan
      -- AND a.level_2_full_code NOT IN ('1804','1805')      -- selain dua terbesar
      -- ==================== OPSI FILTER WILAYAH LAIN ====================
      -- AND a.level_3_full_code = '1871010'
      -- AND a.level_4_full_code = '1871010001'
      -- AND a.level_5_full_code IN ('18710710060045','18710710060047')
      -- ==================== OPSI FILTER LAIN ====================
      -- AND a.assignment_status_id > 0                      -- sudah digarap
      -- AND a.assignment_status_alias LIKE '%APPROVED%'
      -- AND COALESCE(r.geotag_latitude,0) = 0               -- belum geotag saja
),
d AS (
    SELECT CONCAT_WS('^',
               COALESCE(level_2_name, ''),
               COALESCE(level_3_name, ''),
               COALESCE(level_4_name, ''),
               COALESCE(level_6_full_code, ''),
               COALESCE(CAST(geotag_latitude  AS CHAR), ''),
               COALESCE(CAST(geotag_longitude AS CHAR), ''),
               REPLACE(REPLACE(COALESCE(nama_kk, data1, ''), '^', '/'), CHAR(10), ' '),
               REPLACE(REPLACE(COALESCE(nama_ak_lain, ''), '^', '/'), CHAR(10), ' '),
               COALESCE(ada_keluarga_label, ''),
               assignment_id
           ) AS muatan,
           ROW_NUMBER() OVER (ORDER BY level_6_full_code, assignment_id) AS rn
    FROM base
),
b AS (
    SELECT FLOOR((rn - 1) / 24) AS kelompok,
           (rn - 1) % 24        AS slot,
           muatan
    FROM d
)
SELECT kelompok,
       MAX(CASE WHEN slot =  0 THEN muatan END) AS s00,
       MAX(CASE WHEN slot =  1 THEN muatan END) AS s01,
       MAX(CASE WHEN slot =  2 THEN muatan END) AS s02,
       MAX(CASE WHEN slot =  3 THEN muatan END) AS s03,
       MAX(CASE WHEN slot =  4 THEN muatan END) AS s04,
       MAX(CASE WHEN slot =  5 THEN muatan END) AS s05,
       MAX(CASE WHEN slot =  6 THEN muatan END) AS s06,
       MAX(CASE WHEN slot =  7 THEN muatan END) AS s07,
       MAX(CASE WHEN slot =  8 THEN muatan END) AS s08,
       MAX(CASE WHEN slot =  9 THEN muatan END) AS s09,
       MAX(CASE WHEN slot = 10 THEN muatan END) AS s10,
       MAX(CASE WHEN slot = 11 THEN muatan END) AS s11,
       MAX(CASE WHEN slot = 12 THEN muatan END) AS s12,
       MAX(CASE WHEN slot = 13 THEN muatan END) AS s13,
       MAX(CASE WHEN slot = 14 THEN muatan END) AS s14,
       MAX(CASE WHEN slot = 15 THEN muatan END) AS s15,
       MAX(CASE WHEN slot = 16 THEN muatan END) AS s16,
       MAX(CASE WHEN slot = 17 THEN muatan END) AS s17,
       MAX(CASE WHEN slot = 18 THEN muatan END) AS s18,
       MAX(CASE WHEN slot = 19 THEN muatan END) AS s19,
       MAX(CASE WHEN slot = 20 THEN muatan END) AS s20,
       MAX(CASE WHEN slot = 21 THEN muatan END) AS s21,
       MAX(CASE WHEN slot = 22 THEN muatan END) AS s22,
       MAX(CASE WHEN slot = 23 THEN muatan END) AS s23
FROM b
WHERE kelompok >= 0 AND kelompok <= 8999
GROUP BY kelompok
ORDER BY kelompok;
