-- Daftar Assignment Berstatus OPEN / DRAFT
WITH aktif AS (
    SELECT a.assignment_id, a.survey_period_id,
           a.level_2_name, a.level_3_name, a.level_4_name,
           a.level_5_name, a.level_5_full_code,
           a.level_6_name, a.level_6_full_code,
           a.assignment_status_alias, a.current_user_id,
           a.current_user_survey_role_is_pencacah AS cur_is_pcl,
           a.data1, a.data2, a.data3, a.data6,
           TRIM(CASE WHEN LOCATE('/', a.data6) > 0
                     THEN LEFT(a.data6, LOCATE('/', a.data6) - 1)
                     ELSE a.data6 END) AS skala_prelist,
           TRIM(CASE WHEN LOCATE('/', a.data6) > 0
                     THEN SUBSTRING(a.data6, LOCATE('/', a.data6) + 1)
                     ELSE '' END) AS jenis_prelist
    FROM tgr_fd68e454.base_table_assignment a
    WHERE a.is_active = 1
      AND a.assignment_status_alias IN ('OPEN','DRAFT')
),
kel AS (                       -- nama keluarga (bila sudah ada isian)
    SELECT assignment_id, no_kk, nama_kk, dtsen_nama_kk, ada_keluarga_label
    FROM (
        SELECT r.assignment_id, r.no_kk, r.nama_kk, r.dtsen_nama_kk,
               r.ada_keluarga_label,
               ROW_NUMBER() OVER (PARTITION BY r.assignment_id
                                  ORDER BY r.assignment_date_modified DESC) AS rn
        FROM tgr_fd68e454.root_table r
    ) x WHERE rn = 1
),
usa AS (                       -- nama usaha (bila sudah ada isian)
    SELECT assignment_id, nama_usaha, idsbr, keberadaan_usaha_label
    FROM (
        SELECT n.assignment_id, n.nama_usaha, n.idsbr, n.keberadaan_usaha_label,
               ROW_NUMBER() OVER (PARTITION BY n.assignment_id
                                  ORDER BY n.index1) AS rn
        FROM tgr_fd68e454.se2026_nested n
    ) x WHERE rn = 1
),
alok AS (
    SELECT user_id, MAX(email) AS email
    FROM tgr_fd68e454.base_table_user_allocation_new
    GROUP BY user_id
)
SELECT a.level_2_name      AS kabupaten,
       a.level_3_name      AS kecamatan,
       a.level_4_name      AS desa_kel,
       a.level_5_name      AS sls,
       a.level_5_full_code AS id_sls,
       a.level_6_name      AS sub_sls,
       a.level_6_full_code AS id_sub_sls,
       a.assignment_id,
       a.assignment_status_alias AS status,
       CASE WHEN UPPER(a.jenis_prelist) = 'KELUARGA' THEN 'Keluarga'
            WHEN a.jenis_prelist = ''                THEN '(tidak diketahui)'
            ELSE 'Usaha' END AS jenis,
       a.jenis_prelist,
       a.skala_prelist,
       COALESCE(u.nama_usaha, k.dtsen_nama_kk, k.nama_kk, a.data1) AS nama_usaha_atau_keluarga,
       a.data1             AS nama_prelist,
       a.data2             AS alamat_prelist,
       COALESCE(u.idsbr, a.data3) AS idsbr,
       k.no_kk,
       k.ada_keluarga_label       AS keberadaan_keluarga,
       u.keberadaan_usaha_label   AS keberadaan_usaha,
       CASE WHEN a.cur_is_pcl = 1 THEN al.email END AS email_pcl,
       CASE WHEN a.cur_is_pcl = 0 THEN al.email END AS email_pml,
       CONCAT('https://fasih-sm.bps.go.id/app/assignment/',
              a.survey_period_id, '/', a.assignment_id) AS link_fasih
FROM aktif a
LEFT JOIN kel  k  ON k.assignment_id  = a.assignment_id
LEFT JOIN usa  u  ON u.assignment_id  = a.assignment_id
LEFT JOIN alok al ON al.user_id       = a.current_user_id
ORDER BY a.level_6_full_code, a.assignment_status_alias, a.assignment_id
LIMIT 9000 OFFSET 0;
