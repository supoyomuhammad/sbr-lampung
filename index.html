// ── REKAP TARGET ────────────────────────────────────────── function renderRekapTarget() { const tbody = document.getElementById('rekap-tbody'); tbody.innerHTML = ''; let totalSE2016 = 0, totalTarget = 0, totalRealisasi = 0; const tambahan = window._tambahanSejak12 || {}; let rekapData = TARGET_DATA.map(item => { const normId = item.id.replace('TULANGBAWANG','TULANG BAWANG'); const tam = tambahan[normId] || 0; const real = item.base_11_maret + tam; const sisa = item.target - real; const pct = (real / item.se2016) * 100; // Rumus: realisasi dibagi SE2016 x 100 totalSE2016 += item.se2016; totalTarget += item.target; totalRealisasi += real; return {...item, realisasi:real, sisa, pct}; }); // Urutkan dari persentase tertinggi rekapData.sort((a,b) => b.pct - a.pct); rekapData.forEach((row,idx) => { const ok = row.sisa <= 0; const sisaHtml = ok ? `+${Math.abs(row.sisa).toLocaleString('id-ID')} (Lebih)` : `-${row.sisa.toLocaleString('id-ID')}`; const barColor = ok ? 'linear-gradient(90deg,var(--green-light),var(--green))' : 'linear-gradient(90deg,var(--orange-light),var(--orange))'; const pctD = Math.min(row.pct, 100); const tr = document.createElement('tr'); tr.innerHTML = ` ${idx+1} ${row.nama} ${row.se2016.toLocaleString('id-ID')} ${row.target.toLocaleString('id-ID')} ${row.realisasi.toLocaleString('id-ID')}
${row.pct.toFixed(1)}%
${sisaHtml}`; tbody.appendChild(tr); }); const totalSisa = totalTarget - totalRealisasi; const totalPct = (totalRealisasi / totalSE2016) * 100; // Rumus total persentase provinsi const okT = totalSisa <= 0; const trT = document.createElement('tr'); trT.style.background = 'var(--gray-50)'; trT.innerHTML = ` TOTAL PROVINSI LAMPUNG ${totalSE2016.toLocaleString('id-ID')} ${totalTarget.toLocaleString('id-ID')} ${totalRealisasi.toLocaleString('id-ID')}
${totalPct.toFixed(1)}%
${okT?`+${Math.abs(totalSisa).toLocaleString('id-ID')} (Lebih)`:`-${totalSisa.toLocaleString('id-ID')}`}`; tbody.appendChild(trT); }
