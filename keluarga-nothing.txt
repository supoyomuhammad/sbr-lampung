SELECT 
    A.prov, 
    A.kab, 
    A.kec, 
    A.desa, 
    A.kode_sls, 
    A.nama_sls, 
    A.no_kk, 
    A.nama_kk,
    A.nama_ak_lain,
    A.ada_keluarga_label AS status_di_sls_awal,
    'Benar-benar Hilang (Tidak ada di SLS lain)' AS keterangan
    
FROM (
    -- Tabel A: Filter kode kabupaten dimasukkan ke sini agar query lebih cepat
    SELECT prov, kab, kec, desa, kode_sls, nama_sls, no_kk, nama_kk, nama_ak_lain,ada_keluarga_label 
    FROM tgr_fd68e454.root_table 
    WHERE ada_keluarga_value = "0" 
      AND level_2_full_code = '1813' -- GANTI DENGAN KODE KABUPATEN YANG INGIN ANDA TARIK
) AS A 

LEFT JOIN (
    -- Tabel B: Daftar keluarga yang berstatus "Ditemukan" di seluruh database
    SELECT no_kk 
    FROM tgr_fd68e454.root_table 
    WHERE ada_keluarga_value = "1"
) AS B 
ON A.no_kk = B.no_kk

WHERE 
    -- Kunci Anti-Join
    B.no_kk IS NULL

ORDER BY 
    A.prov, 
    A.kab, 
    A.kec, 
    A.desa, 
    A.kode_sls, 
    A.nama_kk

-- PENGATURAN TARIKAN DATA --
LIMIT 5000 OFFSET 0;
